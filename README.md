# Локальный сетевой чат. Один к одному. Очереди сообщений, именованные каналы, сокеты.
# Для сборки 
./build.sh
host_mq      - хост для очередей сообщений
host_fifo    - хост для именованных каналов  
host_sock    - хост для сокетов
client_mq    - клиент для очередей сообщений
client_fifo  - клиент для именованных каналов
client_sock  - клиент для сокетов
# Запуск хоста:
./host_<тип> <тип_соединения> <имя_пользователя>

# Примеры:
./host_mq mq Admin
./host_fifo fifo Admin  
./host_sock sock Admin

# Запуск клиента:
 # Общий формат:
./client_<тип> <тип_соединения> <host_pid> <client_id> <имя_пользователя>

# Примеры:
./client_mq mq <PID_хоста> 1 User1
./client_fifo fifo <PID_хоста> 1 User1
./client_sock sock <PID_хоста> 1 User1

<PID_хоста> - PID процесса хоста, который выводится при его запуске
<client_id> - всегда должен быть 1 для конфигурации "один к одному"
